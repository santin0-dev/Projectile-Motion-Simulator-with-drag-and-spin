FUNCTION advance_epoch(current_state, acceleration):

    get dt from simulator

    separate current position:
        current_x
        current_y

    separate current velocity:
        current_vx
        current_vy

    separate acceleration:
        ax
        ay

    calculate new velocity:
        new_vx = current_vx + ax × dt
        new_vy = current_vy + ay × dt

    calculate new position using new velocity:
        new_x = current_x + new_vx × dt
        new_y = current_y + new_vy × dt

    calculate new time:
        new_time = current_time + dt

    create a NEW State containing:
        time = new_time
        position = (new_x, new_y)
        velocity = (new_vx, new_vy)
        acceleration = (ax, ay)

    return the new State
